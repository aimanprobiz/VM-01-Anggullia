// Toggle promo code group & show website requirement hint when 'code' is selected const actionTypeSelect = document.getElementById('bizActionType'); const promoCodeGroup = document.getElementById('promoCodeGroup'); const websiteInput = document.getElementById('bizWebsite'); const websiteLabel = websiteInput.previousElementSibling; // Grab the label above Website URL actionTypeSelect.addEventListener('change', () => { if (actionTypeSelect.value === 'code') { promoCodeGroup.style.display = 'block'; websiteInput.setAttribute('required', 'true'); websiteLabel.innerHTML = 'Website / Store Link for Promo Code * (Required for customers to use your code)'; } else { promoCodeGroup.style.display = 'none'; websiteInput.removeAttribute('required'); websiteLabel.innerHTML = 'Website URL'; } });
